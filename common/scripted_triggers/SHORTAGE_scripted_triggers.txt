SHORTAGE_governorship_eligible_for_famine = {
	# Scope: Governorship
	# Function: Check if the governorship is eligible to trigger a famine event
	AND = {
		SHORTAGE_food_shortage > SHORTAGE_famine_threshold
		any_governorship_state = {
			any_state_province = {
				SHORTAGE_province_eligible_for_famine = yes
			}
		}
	}
}

SHORTAGE_province_eligible_for_famine = {
	# Scope: Province
	# Function: Check if the province is eligible to trigger a famine event
	total_population > 3
	NOT = {
		has_province_modifier = recent_famine
	}
}
